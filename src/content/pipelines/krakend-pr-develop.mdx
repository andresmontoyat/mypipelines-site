```yaml title="templates/krakend-pr-develop.yml"
# Template: PR to develop validation
# Copy to: .github/workflows/pr-develop.yml
#
# Triggers on pull requests targeting develop branch
# Runs: commit-lint + config validation + audit

name: PR Validation

on:
  pull_request:
    branches:
      - develop

# A newer push to the same PR or branch makes this run's answer stale.
concurrency:
  group: ${{ github.workflow }}-${{ github.event.pull_request.number || github.ref }}
  cancel-in-progress: true

jobs:
  validate:
    uses: <org>/ci-templates/.github/workflows/krakend-main-pipeline.yml@v2
    with:
      run_commit_lint: true
      run_build: true
      run_test: true
      audit_threshold: 0
    secrets: inherit
```
